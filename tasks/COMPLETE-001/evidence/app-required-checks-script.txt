import hashlib,datetime
import os,json,subprocess,urllib.request,urllib.error
from pathlib import Path
c=subprocess.run(['git','-c','credential.interactive=never','credential','fill'],input='protocol=https\nhost=github.com\n\n',capture_output=True,text=True,check=True)
token=next(s[9:] for s in c.stdout.splitlines() if s.startswith('password='))
class NoRedirect(urllib.request.HTTPRedirectHandler):
 def redirect_request(self,*a,**k):return None
opener=urllib.request.build_opener(NoRedirect())
def api(path,method='GET',payload=None):
 req=urllib.request.Request('https://api.github.com/'+path,method=method,data=None if payload is None else json.dumps(payload).encode(),headers={'Authorization':'Bearer '+token,'Content-Type':'application/json','Accept':'application/vnd.github+json','X-GitHub-Api-Version':'2022-11-28','User-Agent':'foundation-completion'})
 with opener.open(req,timeout=30) as r:
  b=r.read();return json.loads(b) if b else None
plan=json.loads(Path('tasks/COMPLETE-001/evidence/app-required-checks-plan.json').read_text(encoding='utf-8'))
slug='repos/trdat1605/s-251521f7e9131fb0/'
repo=api(slug[:-1]);assert repo['id']==1467945011 and repo['default_branch']=='main'
assert api(slug+'git/ref/heads/main')['object']['sha']==plan['expected_main']
assert api(slug+'git/ref/heads/codex/portable-foundation')['object']['sha']==plan['expected_head']
current=api(slug+'rulesets/27859157')
selected={k:current[k] for k in ['name','target','enforcement','bypass_actors','conditions','rules']};assert selected==plan['before'],'Ruleset changed since plan'
candidate=json.loads(Path(plan['candidate_file']).read_text(encoding='utf-8'));assert candidate==plan['candidate'];assert hashlib.sha256(Path(plan['candidate_file']).read_bytes()).hexdigest()==plan['candidate_file_sha256']
base={k:v for k,v in candidate.items() if k!='rules'};assert base=={k:v for k,v in selected.items() if k!='rules'}
assert [x for x in candidate['rules'] if x['type']!='required_status_checks']==selected['rules']
assert {x['context'] for x in candidate['rules'][-1]['parameters']['required_status_checks']}=={'metadata','focused-frontend','isolated-runtime'}
checks=api(slug+'commits/'+plan['expected_head']+'/check-runs')['check_runs']
for name in ['metadata','focused-frontend','isolated-runtime']:
 assert any(x['name']==name and x['app']['id']==15368 and x['status']=='completed' and x['conclusion']=='success' for x in checks)
# Only this owned ruleset receives the three observed required checks. No bypass or heads mutation.
api(slug+'rulesets/27859157','PUT',candidate)
after=api(slug+'rulesets/27859157');after={k:after[k] for k in ['name','target','enforcement','bypass_actors','conditions','rules']};assert after==candidate
assert api(slug+'git/ref/heads/main')['object']['sha']==plan['expected_main']
effective=api(slug+'rules/branches/main')
assert all(any(r['type']=='required_status_checks' and r['parameters']==candidate['rules'][-1]['parameters'] for r in effective) for _ in [0])
out={'task_id':'COMPLETE-001','status':'PASS','exit_code':0,'operation':'Update owned app ruleset27859157 to require the three actual successful CI jobs; no bypass/main/runtime changes','before':selected,'after':after,'main_unchanged':plan['expected_main'],'app_head':plan['expected_head'],'run_id':plan['run_id'],'effective_rule_types':[r['type'] for r in effective],'observed_utc':datetime.datetime.now(datetime.timezone.utc).isoformat(),'limits':'Server config/readback, not actual merge or fullproduct acceptance'}
Path('tasks/COMPLETE-001/evidence/app-required-checks-applied.json').write_text(json.dumps(out,indent=2)+'\n',encoding='utf-8',newline='\n')
print('PASS: three app required checks enforced; main unchanged.')
